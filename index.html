function processData(csvText) { try { const rows = csvText.split('\n'); globalData = []; for (let i = 1; i < rows.length; i++) { const row = rows[i].trim(); if(!row) continue; // 升級版 CSV 解析器：完美處理空格與引號 let cols = []; let currentVal = ''; let inQuotes = false; for (let j = 0; j < row.length; j++) { let char = row[j]; if (char === '"') { inQuotes = !inQuotes; // 遇到引號切換狀態 } else if (char === ',' && !inQuotes) { cols.push(currentVal.trim()); // 不在引號內的逗號才是分隔符 currentVal = ''; } else { currentVal += char; } } cols.push(currentVal.trim()); // 推入最後一個欄位 if (cols.length >= 3) { globalData.push({ cat: (cols[0] || '未分類').replace(/"/g, ''), zh: (cols[1] || '').replace(/"/g, ''), jp: (cols[2] || '').replace(/"/g, ''), romaji: (cols[3] || '').replace(/"/g, '') }); } } renderCategories(); filterData(); document.getElementById('loading').style.display = 'none'; } catch(e) { showError("資料解析失敗: " + e.message); } }
